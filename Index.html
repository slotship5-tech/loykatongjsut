📊 เชื่อมต่อฐานข้อมูลออนไลน์ Google Sheets (ลอยกระทงร่วมกันทุกเครื่อง)

วิธีติดตั้ง Google Sheets ฟรี ภายใน 2 นาที:

  1. สร้าง Google Sheet ใหม่ > ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  2. วางโค้ดด้านล่างนี้ แล้วกด ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้รายการใหม่ (New deployment)
  3. เลือกประเภทเป็น เว็บแอป (Web app) > ตั้งค่าผู้มีสิทธิ์เข้าถึงเป็น ทุกคน (Anyone) แล้วคัดลอก URL มาวางด้านบน
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var list = [];
  for (var i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    list.push({
      id: String(rows[i][0]), name: String(rows[i][1]), wish: String(rows[i][2]),
      base: String(rows[i][3] || 'darathong'), flower: String(rows[i][4] || 'lotus'),
      candle: String(rows[i][5] || 'yellow'), aura: String(rows[i][6] || 'goldsparkle'),
      likes: Number(rows[i][7] || 1), timestamp: String(rows[i][8] || '')
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', items: list.reverse() })).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) sheet.appendRow(['id','name','wish','base','flower','candle','aura','likes','timestamp']);
  var data = JSON.parse(e.postData.contents);
  if (data.action === 'add') {
    var k = data.item;
    sheet.appendRow([k.id, k.name, k.wish, k.base, k.flower, k.candle, k.aura, k.likes || 1, k.timestamp || '']);
  } else if (data.action === 'like' || data.action === 'edit' || data.action === 'delete') {
    var rows = sheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(data.id)) {
        if (data.action === 'like') sheet.getRange(i + 1, 8).setValue(data.likes);
        if (data.action === 'edit') { sheet.getRange(i + 1, 2).setValue(data.name); sheet.getRange(i + 1, 3).setValue(data.wish); }
        if (data.action === 'delete') sheet.deleteRow(i + 1);
        break;
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' })).setMimeType(ContentService.MimeType.JSON);
}
